# omo V2 P2 — Tool 注入 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 omo 的 V1 工具接入 OpenCode V2，使模型可用（带命名空间、权限、结果规范化、context shim）。

**Architecture:** 复用 omo 的 `createTools`（产出 V1 `ToolsRecord`），用纯函数 `toV2Tool` 把每个 V1 `ToolDefinition` 翻译成 V2 `Info`（`input` 直接传 `z.object(args)`；`execute` 包装 V1 executor 并 shim context；结果规范化为 `{content}`），在 `setup(ctx)` 中经 `ctx.tool.transform` 注册（namespace + `codemode:false` + 幂等）。

**Tech Stack:** TypeScript、Bun/`bun:test`、`tsgo`、`@opencode/plugin@2.0.11`、zod 4.4.3（StandardSchemaV1）。

## Global Constraints

- V2 API 仅 `@opencode/plugin@^2.0.11`；V1 包不变。
- **`input` 直接传 `z.object(def.args)`**（spike 已证 zod 可被接受）；不要做 JSON Schema 转换。
- 工具必须 `namespace`（V2 自带 `glob`/`grep`/`skill`）并 `codemode:false`。
- **`editor.add` 必须幂等**（plugin `setup` 可能多次运行）。
- `directory` 按调用解析（V2 无 per-call directory；静态回退到 `ctx.location.directory` 并在注释标注 session-move 限制）。
- V2 `ToolContext` 无 `signal`/`ask`/`metadata`/`worktree` → shim 提供可用的近似或显式降级；不得静默丢弃 V1 `ToolResult.title`/`attachments`。
- 不修改用户全局配置；验证放 `/tmp/omo-v2-tool-probe`（软链了 fork 的 node_modules）。
- SUL-1.0：不得移除许可证/版权声明。
- 仓库 `/home/user/omo-v2`；`export PATH="$HOME/.bun/bin:$PATH"`；网络需代理 `export https_proxy=http://192.168.170.73:7898 http_proxy=http://192.168.170.73:7898`。

// packages/omo-opencode/src/v2/tool-mapping.ts
import { z } from "zod"

export type V1ToolDefinition = {
  description: string
  args: z.ZodRawShape
  execute: (args: any, context: any) => Promise<unknown>
}

type V2Context = {
  sessionID: string
  agent: string
  messageID: string
  id: string
  progress: (update: Record<string, unknown>) => Promise<void> | unknown
}

type V2Content = { type: "text"; text: string } | { type: "file"; uri: string; mime: string; name?: string }

export function normalizeToolResult(result: unknown): {
  content: string | V2Content[]
  metadata?: Record<string, unknown>
} {
  if (typeof result === "string") return { content: result }
  if (typeof result === "object" && result !== null) {
    const r = result as {
      title?: string
      output?: string
      metadata?: Record<string, unknown>
      attachments?: Array<{ type: "file"; mime: string; url: string; filename?: string }>
    }
    const metadata = { ...(r.title !== undefined ? { title: r.title } : {}), ...(r.metadata ?? {}) }
    const base = r.output ?? ""
    const attachments = r.attachments ?? []
    const content: string | V2Content[] =
      attachments.length === 0
        ? base
        : [
            { type: "text", text: base },
            ...attachments.map((a) => ({
              type: "file" as const,
              uri: a.url,
              mime: a.mime,
              ...(a.filename !== undefined ? { name: a.filename } : {}),
            })),
          ]
    return Object.keys(metadata).length > 0 ? { content, metadata } : { content }
  }
  return { content: String(result) }
}

/**
 * V2 ToolContext has no `directory`, `worktree`, `abort`, `ask`, or `metadata`.
 * - directory/worktree: static fallback to the plugin location (session moves are NOT reflected; see P2 findings).
 * - abort: a non-aborting signal is supplied because V2 exposes none.
 * - ask: a resolver is not available here; the shim forwards to progress and resolves.
 */
export function toV2Tool(
  name: string,
  def: V1ToolDefinition,
  opts: { namespace: string; directory: string },
): {
  name: string
  description: string
  input: z.ZodObject<z.ZodRawShape>
  execute: (input: unknown, context: V2Context) => Promise<{ content: string | V2Content[]; metadata?: Record<string, unknown> }>
  options: { namespace: string; codemode: false }
} {
  const controller = new AbortController()
  return {
    name,
    description: def.description,
    input: z.object(def.args),
    options: { namespace: opts.namespace, codemode: false },
    execute: async (input, context) => {
      const shimmed = {
        sessionID: context.sessionID,
        messageID: context.messageID,
        agent: context.agent,
        directory: opts.directory,
        worktree: opts.directory,
        abort: controller.signal,
        metadata: (i: { title?: string; metadata?: Record<string, unknown> }) => {
          void context.progress({ ...(i.metadata ?? {}), ...(i.title !== undefined ? { title: i.title } : {}) })
        },
        ask: async () => {},
      }
      return normalizeToolResult(await def.execute(input, shimmed))
    },
  }
}
```

// packages/omo-opencode/src/v2/register-tools.ts
import type { Plugin } from "@opencode/plugin"
import { createManagers } from "../create-managers"
import { createTools } from "../create-tools"
import { createRuntimeTmuxConfig } from "../create-runtime-tmux-config"
import { createModelCacheState } from "../plugin-state"
import { log } from "../shared"
import { toV2Tool } from "./tool-mapping"

type V2Ctx = Parameters<NonNullable<Parameters<typeof Plugin.define>[0]["setup"]>>[0]

const NAMESPACE = "omo"

export async function registerOmoTools(ctx: V2Ctx, pluginConfig: Record<string, unknown>): Promise<void> {
  const directory = ctx.location.directory
  const v1Ctx = { directory } as never
  const tmuxConfig = createRuntimeTmuxConfig(pluginConfig as never)
  const modelCacheState = createModelCacheState()
  const managers = createManagers({
    ctx: v1Ctx,
    pluginConfig: pluginConfig as never,
    tmuxConfig,
    modelCacheState,
    backgroundNotificationHookEnabled: false,
  })
  const { filteredTools } = await createTools({ ctx: v1Ctx, pluginConfig: pluginConfig as never, managers })

  const entries = Object.entries(filteredTools)
  const built = entries.map(([name, def]) =>
    toV2Tool(
      name,
      def as unknown as { description: string; args: never; execute: (a: unknown, c: unknown) => Promise<unknown> },
      { namespace: NAMESPACE, directory },
    ),
  )

  await ctx.tool.transform((editor) => {
    editor.namespace({ name: NAMESPACE, description: "oh-my-openagent tools" })
    for (const tool of built) {
      // idempotent: update only if absent, since setup may run more than once
      const existing = editor.get(`${NAMESPACE}_${tool.name}`)
      if (existing === undefined) editor.add(tool as never)
      else editor.update(`${NAMESPACE}_${tool.name}`, (t) => Object.assign(t, tool as never))
    }
  })

  log("[v2] tools registered", { toolCount: entries.length })
}
```
